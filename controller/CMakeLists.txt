cmake_minimum_required(VERSION 3.16)
project(g1_sim_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(UPSTREAM "${REPO}/.deps/src/unitree_rl_mjlab/deploy")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${REPO}/.deps/src/unitree_sdk2 sdk2)
find_package(Boost REQUIRED COMPONENTS program_options)
find_package(yaml-cpp REQUIRED)
find_package(ZLIB REQUIRED)
find_package(fmt REQUIRED)
set(ORT "${UPSTREAM}/thirdparty/onnxruntime-linux-x64-1.22.0")
include_directories(/usr/include/eigen3 ${UPSTREAM}/include ${UPSTREAM}/robots/g1/include ${ORT}/include ${UPSTREAM}/thirdparty/cnpy)
add_executable(g1_ctrl main.cpp ${UPSTREAM}/robots/g1/src/State_RLBase.cpp ${UPSTREAM}/thirdparty/cnpy/cnpy.cpp)
target_link_libraries(g1_ctrl unitree_sdk2 Boost::program_options yaml-cpp fmt::fmt ZLIB::ZLIB ${ORT}/lib/libonnxruntime.so.1.22.0)
add_library(g1_dds_bridge SHARED dds_bridge.cpp)
target_link_libraries(g1_dds_bridge unitree_sdk2)
